cmake_minimum_required(VERSION 3.16)
project(rknn_bridge_experiment LANGUAGES C)
set(BRIDGE_REFERENCE_SOURCE "" CACHE FILEPATH "Original deployed bridge source")
set(RKNN_HEADERS "" CACHE PATH "Original SDK header directory")
set(RKNN_LIBRARY "" CACHE FILEPATH "Installed librknnrt.so")
foreach(path "${BRIDGE_REFERENCE_SOURCE}" "${RKNN_HEADERS}/rknn_api.h" "${RKNN_LIBRARY}")
    if(NOT EXISTS "${path}")
        message(FATAL_ERROR "Required existing RKNN artifact missing: ${path}")
    endif()
endforeach()
foreach(target rknn_profile_reference rknn_output_prealloc)
    add_library(${target} SHARED bridge.c)
    target_compile_features(${target} PRIVATE c_std_11)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -fno-fast-math)
    target_include_directories(${target} PRIVATE "${RKNN_HEADERS}")
    target_compile_definitions(${target} PRIVATE BRIDGE_REFERENCE_SOURCE="${BRIDGE_REFERENCE_SOURCE}")
    target_link_libraries(${target} PRIVATE "${RKNN_LIBRARY}")
endforeach()
target_compile_definitions(rknn_output_prealloc PRIVATE OUTPUT_PREALLOC=1)
