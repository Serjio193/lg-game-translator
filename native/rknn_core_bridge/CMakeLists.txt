cmake_minimum_required(VERSION 3.16)
project(rknn_core_bridge LANGUAGES C)
set(BRIDGE_REFERENCE_SOURCE "" CACHE FILEPATH "Original deployed bridge source")
set(RKNN_HEADERS "" CACHE PATH "Original SDK header directory")
set(RKNN_LIBRARY "" CACHE FILEPATH "Installed librknnrt.so")
foreach(path "${BRIDGE_REFERENCE_SOURCE}" "${RKNN_HEADERS}/rknn_api.h" "${RKNN_LIBRARY}")
    if(NOT EXISTS "${path}")
        message(FATAL_ERROR "Required existing RKNN artifact missing: ${path}")
    endif()
endforeach()
add_library(rknn_core_bridge SHARED bridge.c)
target_compile_features(rknn_core_bridge PRIVATE c_std_11)
target_compile_options(rknn_core_bridge PRIVATE -Wall -Wextra -Werror -fno-fast-math)
target_include_directories(rknn_core_bridge PRIVATE "${RKNN_HEADERS}")
target_compile_definitions(rknn_core_bridge PRIVATE BRIDGE_REFERENCE_SOURCE="${BRIDGE_REFERENCE_SOURCE}")
target_link_libraries(rknn_core_bridge PRIVATE "${RKNN_LIBRARY}")
