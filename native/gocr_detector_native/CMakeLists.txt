cmake_minimum_required(VERSION 3.16)
project(gocr_detector_native LANGUAGES CXX)
add_subdirectory(../gocr_postprocess postprocess)
add_library(gocr_detector_native SHARED gocr_detector.cpp tflite_runtime.cpp preprocess.cpp decode.cpp profiler.cpp)
target_compile_features(gocr_detector_native PRIVATE cxx_std_17)
target_compile_options(gocr_detector_native PRIVATE -O2 -Wall -Wextra -Werror
    -fno-fast-math -ffp-contract=off)
target_include_directories(gocr_detector_native PUBLIC . ../gocr_postprocess)
target_link_libraries(gocr_detector_native PRIVATE gocr_postprocess dl pthread)
set_target_properties(gocr_detector_native PROPERTIES BUILD_RPATH "$ORIGIN" INSTALL_RPATH "$ORIGIN")
add_executable(gocr_detector_bench benchmark.cpp)
target_compile_features(gocr_detector_bench PRIVATE cxx_std_17)
target_compile_options(gocr_detector_bench PRIVATE -O2 -Wall -Wextra -Werror)
target_link_libraries(gocr_detector_bench PRIVATE gocr_detector_native)
set_target_properties(gocr_detector_bench PROPERTIES BUILD_RPATH "$ORIGIN" INSTALL_RPATH "$ORIGIN/../lib")
install(TARGETS gocr_detector_native gocr_detector_bench LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
